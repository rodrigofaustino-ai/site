// Acrescente DENTRO de match /databases/{database}/documents.
// Preserve todas as regras existentes. Somente no projeto salamulti.
match /alertasFrequenciaConfig/{uid} {
  function donoConfiguracao() {
    return request.auth != null && request.auth.uid == uid;
  }
  function configuracaoValida() {
    let d = request.resource.data;
    return d.keys().hasAll(['dias','seguidas','alternadas','atualizadoEm'])
      && d.keys().hasOnly(['dias','seguidas','alternadas','atualizadoEm'])
      && d.dias is int && d.dias >= 1 && d.dias <= 3650
      && d.seguidas is int && d.seguidas >= 1 && d.seguidas <= 365
      && d.alternadas is int && d.alternadas >= 1 && d.alternadas <= 365
      && d.atualizadoEm == request.time;
  }
  allow get: if donoConfiguracao();
  allow list: if false;
  allow create, update: if donoConfiguracao() && configuracaoValida();
  allow delete: if false;
}
